FROM debian:bookworm-slim AS build
RUN apt-get update && apt-get install -y --no-install-recommends g++ && rm -rf /var/lib/apt/lists/*
WORKDIR /src
COPY server.cpp .
RUN g++ -std=c++20 -O2 server.cpp -o server
FROM debian:bookworm-slim
COPY --from=build /src/server /usr/local/bin/harbor-dispatch
USER 10001
ENTRYPOINT ["/usr/local/bin/harbor-dispatch"]
CMD ["--container"]
