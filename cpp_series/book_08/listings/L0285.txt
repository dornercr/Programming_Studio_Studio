terraform { required_version = ">= 1.4.0" }
variable "replicas" {
  type    = number
  default = 3
  validation {
    condition     = var.replicas >= 2 && floor(var.replicas) == var.replicas
    error_message = "replicas must be an integer of at least two."
  }
}
resource "terraform_data" "service_contract" {
  input = { replicas = var.replicas, port = 8080 }
}
output "contract" { value = terraform_data.service_contract.output }
