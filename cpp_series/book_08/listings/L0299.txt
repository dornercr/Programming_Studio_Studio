apiVersion: apps/v1
kind: Deployment
metadata: {name: harbor-zones}
spec:
  replicas: 3
  selector: {matchLabels: {app: harbor-zones}}
  template:
    metadata: {labels: {app: harbor-zones}}
    spec:
      topologySpreadConstraints:
        - maxSkew: 1
          topologyKey: topology.kubernetes.io/zone
          whenUnsatisfiable: DoNotSchedule
          labelSelector: {matchLabels: {app: harbor-zones}}
      containers:
        - name: app
          image: harbor-service:lab
          imagePullPolicy: Never
          resources:
            requests: {cpu: 100m, memory: 64Mi}
