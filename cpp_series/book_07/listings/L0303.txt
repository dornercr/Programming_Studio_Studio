cmake_minimum_required(VERSION 3.20)
project(Section28Lab LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(support upstream/src/net.cpp upstream/src/store.cpp)
target_include_directories(support PUBLIC upstream/include)
target_compile_features(support PUBLIC cxx_std_20)
target_link_libraries(support PUBLIC Threads::Threads SQLite::SQLite3)
foreach(name client gateway worker)
 add_executable(harbor_${name} upstream/src/${name}.cpp)
 target_link_libraries(harbor_${name} PRIVATE support)
endforeach()
